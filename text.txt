//////////////////////
// MULTI TENANCY
//////////////////////

model Workspace {
  id        String   @id @default(uuid())
  name      String
  createdAt DateTime @default(now())
  users                User[]
  audiences            Audience[]
  contacts             AudienceContact[]
  campaigns            Campaign[]
  messages             CampaignMessage[]
  templates            Template[]
  numbers              SenderNumber[]
  chatwootConversation ChatwootConversation[]
  webHooks             WebhookEvent[]
}

model User {
  id          String @id @default(uuid())
  name        String
  email       String @unique
  workspaceId String

  workspace Workspace @relation(fields: [workspaceId], references: [id])

  campaigns Campaign[]
}

//////////////////////
// TEMPLATE
//////////////////////

model Template {
  id          String @id @default(uuid())
  workspaceId String

  name      String
  content   String
  channel   Channel
  variables Json?

  createdAt DateTime @default(now())

  workspace Workspace  @relation(fields: [workspaceId], references: [id])
  campaigns Campaign[]
}

//////////////////////
// AUDIENCE + CONTACTS
//////////////////////

model Audience {
  id          String @id @default(uuid())
  workspaceId String

  name          String
  totalContacts Int    @default(0)

  createdAt DateTime @default(now())

  workspace Workspace @relation(fields: [workspaceId], references: [id])

  contacts  AudienceContact[]
  campaigns Campaign[]
}

model AudienceContact {
  id         String  @id @default(uuid())
  audienceId String
  sellerId   String?

  name String?

  phoneRaw        String
  phoneNormalized String

  email String?

  hasWhatsapp Boolean @default(true)
  isBlocked   Boolean @default(false)

  chatwootContactId String?
  chatwootSourceId  String?
  workspaceId       String

  createdAt DateTime @default(now())

  audience  Audience          @relation(fields: [audienceId], references: [id], onDelete: Cascade)
  seller    Seller?           @relation(fields: [sellerId], references: [id])
  workspace Workspace?        @relation(fields: [workspaceId], references: [id])
  messages  CampaignMessage[]

  @@unique([workspaceId, phoneNormalized])
  @@index([audienceId])
  @@index([phoneNormalized])
  @@index([chatwootContactId])
}

//////////////////////
// SELLER
//////////////////////

model Seller {
  id    String  @id @default(uuid())
  name  String
  phone String?

  chatwootAgentId Int?

  createdAt DateTime @default(now())

  contacts AudienceContact[]
}

//////////////////////
// CAMPAIGN
//////////////////////

model Campaign {
  id          String @id @default(uuid())
  workspaceId String

  name       String
  userId     String
  templateId String
  audienceId String

  status CampaignStatus @default(DRAFT)

  scheduledAt DateTime?
  startedAt   DateTime?
  finishedAt  DateTime?

  totalContacts     Int?
  processedContacts Int?

  createdAt DateTime @default(now())

  workspace Workspace @relation(fields: [workspaceId], references: [id])
  user      User      @relation(fields: [userId], references: [id])
  template  Template  @relation(fields: [templateId], references: [id])
  audience  Audience  @relation(fields: [audienceId], references: [id])

  messages CampaignMessage[]
  batches  CampaignBatch[]
  stats    CampaignStats?

  @@index([status])
  @@index([scheduledAt])
}

//////////////////////
// CAMPAIGN MESSAGE
//////////////////////

model CampaignMessage {
  id String @id @default(uuid())

  campaignId String
  contactId  String

  phoneNormalized String

  batchId String?

  senderNumberId String?

  status MessageStatus @default(PENDING)

  error String?

  providerId String?

  chatwootConversationId Int?
  chatwootInboxId        Int?

  lastEventType MessageEventType?

  queuedAt    DateTime?
  sentAt      DateTime?
  deliveredAt DateTime?
  readAt      DateTime?
  lockedAt    DateTime?

  workspaceId String

  retryCount Int @default(0)

  createdAt DateTime @default(now())

  campaign     Campaign        @relation(fields: [campaignId], references: [id], onDelete: Cascade)
  contact      AudienceContact @relation(fields: [contactId], references: [id])
  batch        CampaignBatch?  @relation(fields: [batchId], references: [id])
  senderNumber SenderNumber?   @relation(fields: [senderNumberId], references: [id])
  workspace    Workspace       @relation(fields: [workspaceId], references: [id])
  events       MessageEvent[]

  //@@unique([campaignId, contactId]) 
  @@index([campaignId])
  @@index([campaignId, status])
  @@index([status, retryCount])
  @@index([phoneNormalized])
  @@index([providerId])
  @@index([chatwootConversationId])
}

//////////////////////
// BATCH
//////////////////////

model CampaignBatch {
  id String @id @default(uuid())

  campaignId String

  batchIndex Int
  size       Int

  status BatchStatus @default(PENDING)

  startedAt  DateTime?
  finishedAt DateTime?

  createdAt DateTime          @default(now())
  messages  CampaignMessage[]

  campaign Campaign @relation(fields: [campaignId], references: [id], onDelete: Cascade)

  @@index([campaignId])
}

//////////////////////
// STATS
//////////////////////

model CampaignStats {
  id String @id @default(uuid())

  campaignId String @unique

  totalMessages     Int @default(0)
  sentMessages      Int @default(0)
  deliveredMessages Int @default(0)
  readMessages      Int @default(0)
  failedMessages    Int @default(0)

  updatedAt DateTime @updatedAt

  campaign Campaign @relation(fields: [campaignId], references: [id], onDelete: Cascade)
}

// CHATWOOT

model ChatwootConversation {
  id String @id @default(uuid())

  chatwootConversationId Int @unique
  chatwootContactId      Int

  inboxId     Int
  workspaceId String
  contactId   String?

  lastMessageAt DateTime?

  createdAt DateTime @default(now())

  workspace Workspace @relation(fields: [workspaceId], references: [id])

  @@index([chatwootContactId])
}

//////////////////////
// EVENTS (AUDITORIA)
//////////////////////

model MessageEvent {
  id String @id @default(uuid())

  messageId String
  type      MessageEventType

  providerPayload Json?

  createdAt DateTime @default(now())

  message CampaignMessage @relation(fields: [messageId], references: [id], onDelete: Cascade)

  @@index([messageId])
  @@index([type])
}

//////////////////////
// SENDER NUMBER (POOL)
//////////////////////

model SenderNumber {
  id          String @id @default(uuid())
  workspaceId String

  number String @unique

  status String // ACTIVE, BLOCKED, WARMING

  qualityScore Float @default(1.0)

  maxPerSecond Int @default(20)

  lockedAt DateTime?

  createdAt DateTime @default(now())

  workspace Workspace @relation(fields: [workspaceId], references: [id])

  messages CampaignMessage[]
}

//////////////////////
// WEBHOOK
//////////////////////

model WebhookEvent {
  id String @id @default(uuid())

  externalId  String?         @unique
  workspaceId String
  provider    WebhookProvider
  eventType   String

  payload Json

  processed Boolean @default(false)

  createdAt DateTime @default(now())

  workspace Workspace @relation(fields: [workspaceId], references: [id])
}
